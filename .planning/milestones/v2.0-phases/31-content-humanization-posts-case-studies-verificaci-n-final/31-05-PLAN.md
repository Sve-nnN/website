Safe in-place rewrite pattern (walk the tree, only replace `text` values on genuine prose leaves, everything else passes through untouched):
```typescript
type LexicalNode = {
  type: string
  text?: string
  children?: LexicalNode[]
  fields?: Record<string, unknown>
  [key: string]: unknown
}

function rewriteProse(
  node: LexicalNode,
  parentType: string | null,
  rewrite: (original: string, context: 'heading' | 'paragraph' | 'listitem') => string,
): LexicalNode {
  // Never touch code-block embeds or table structure/data.
  if (node.type === 'block' || node.type === 'table') return node
  if (node.type === 'text' && typeof node.text === 'string') {
    if (parentType === 'heading') return { ...node, text: rewrite(node.text, 'heading') }
    if (parentType === 'paragraph') return { ...node, text: rewrite(node.text, 'paragraph') }
    if (parentType === 'listitem') return { ...node, text: rewrite(node.text, 'listitem') }
    return node // e.g. parentType === 'link' — anchor label stays as-is (proper noun/tool name)
  }
  if (Array.isArray(node.children)) {
    return { ...node, children: node.children.map((child) => rewriteProse(child, node.type, rewrite)) }
  }
  return node
}
```
