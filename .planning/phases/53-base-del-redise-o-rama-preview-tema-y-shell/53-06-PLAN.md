```
PW="$(openssl rand -hex 24)"
sudo docker exec -i shared-postgres psql -U postgres -v ON_ERROR_STOP=1 -v pw="$PW" <<'SQL'
CREATE ROLE juantech_preview_ro WITH LOGIN ENCRYPTED PASSWORD :'pw' NOSUPERUSER NOCREATEDB NOCREATEROLE CONNECTION LIMIT 8;
ALTER ROLE juantech_preview_ro SET default_transaction_read_only = on;
GRANT CONNECT ON DATABASE juantech TO juantech_preview_ro;
\c juantech
GRANT USAGE ON SCHEMA public TO juantech_preview_ro;
GRANT SELECT ON ALL TABLES IN SCHEMA public TO juantech_preview_ro;
ALTER DEFAULT PRIVILEGES FOR ROLE juantech_user IN SCHEMA public GRANT SELECT ON TABLES TO juantech_preview_ro;
DO $do$
DECLARE t text;
BEGIN
  FOREACH t IN ARRAY ARRAY['users','users_sessions','subscribers','payload_mcp_api_keys','payload_preferences','payload_preferences_rels'] LOOP
    IF to_regclass('public.' || t) IS NOT NULL THEN
      EXECUTE format('REVOKE SELECT ON public.%I FROM juantech_preview_ro', t);
    END IF;
  END LOOP;
END
$do$;
SQL
echo "DATABASE_URI=postgresql://juantech_preview_ro:${PW}@shared-postgres:5432/juantech"
```

```
sudo docker exec -e PGPASSWORD="$PW" -i shared-postgres psql -h 127.0.0.1 -U juantech_preview_ro -d juantech -c "INSERT INTO pages DEFAULT VALUES;"
sudo docker exec -e PGPASSWORD="$PW" -i shared-postgres psql -h 127.0.0.1 -U juantech_preview_ro -d juantech -c "SELECT count(*) FROM users;"
```

Esperado: el primero devuelve un conteo; el segundo falla con `permission denied` o con un mensaje de transacción de solo lectura; el tercero falla con `permission denied for table users`. Si `-h 127.0.0.1` no conecta, prueba con `-h localhost` y dime qué pasó.
