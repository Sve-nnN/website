function extractJsonLd(html) {
  const matches = [...html.matchAll(/<script type="application\/ld\+json">([\s\S]*?)<\/script>/gi)]
  return matches.map((m) => { const raw = JSON.parse(m[1]); return { type: raw['@type'] ?? null, raw } })
}
async function captureUrl(baseUrl, path) {
  const res = await fetch(`${baseUrl}${path}`)
  if (!res.ok) throw new Error(`GET ... returned ${res.status}`)
  const html = await res.text()
  return { h1: extractH1s(html), jsonLd: extractJsonLd(html) }
}
