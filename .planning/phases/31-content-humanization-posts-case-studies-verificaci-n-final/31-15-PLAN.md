Simple flat-paragraph rewrite pattern (safe here specifically because these 2 fields contain ONLY paragraph/text nodes — do not reuse this simplified full-rebuild approach for Posts.content, which has much richer structure):
```typescript
function paragraph(text: string) {
  return { type: 'paragraph', version: 1, children: [{ type: 'text', version: 1, text }], direction: 'ltr', format: '', indent: 0 }
}
function richTextFromParagraphs(paragraphs: string[]) {
  return {
    root: {
      type: 'root',
      children: paragraphs.map(paragraph),
      direction: 'ltr', format: '', indent: 0, version: 1,
    },
  }
}
```
